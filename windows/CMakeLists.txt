cmake_minimum_required(VERSION 3.20)
project(HandoffVibeWindows VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)

if(WIN32)
    add_executable(handoff-vibe WIN32 window.cpp)
    add_executable(handoff-vibe-diagnostics main.cpp)
    target_link_libraries(handoff-vibe PRIVATE comctl32 uxtheme shell32)
    target_link_libraries(handoff-vibe-diagnostics PRIVATE ws2_32 bthprops)
    foreach(app handoff-vibe handoff-vibe-diagnostics)
        target_compile_definitions(${app} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE _WIN32_WINNT=0x0A00)
        if(MSVC)
            target_compile_options(${app} PRIVATE /W4 /WX /utf-8)
            set_property(TARGET ${app} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
        endif()
    endforeach()
    if(MSVC)
        target_link_options(handoff-vibe PRIVATE
            "/MANIFESTDEPENDENCY:type='win32' name='Microsoft.Windows.Common-Controls' version='6.0.0.0' processorArchitecture='*' publicKeyToken='6595b64144ccf1df' language='*'")
    elseif(MINGW)
        target_link_options(handoff-vibe PRIVATE -municode)
    endif()
    if(BUILD_TESTING)
        add_test(NAME cli-version COMMAND handoff-vibe-diagnostics --version)
        set_tests_properties(cli-version PROPERTIES PASS_REGULAR_EXPRESSION "0[.]1[.]0-beta[.]1")
        add_test(NAME cli-help COMMAND handoff-vibe-diagnostics --help)
        set_tests_properties(cli-help PROPERTIES PASS_REGULAR_EXPRESSION "probe")
        add_test(NAME cli-invalid-address COMMAND handoff-vibe-diagnostics probe 00:00:00:00:00:00)
        set_tests_properties(cli-invalid-address PROPERTIES WILL_FAIL TRUE)
    endif()
endif()

if(BUILD_TESTING)
    add_executable(test-address tests/address.cpp)
    target_include_directories(test-address PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    add_test(NAME address COMMAND test-address)
endif()
